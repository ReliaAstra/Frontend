    39→  { value: 60, label: "60s" },
    40→  { value: 300, label: "5m" },

    50→
    51→type StatusBadgeType = "operational" | "degraded" | "down" | "unknown";
    52→
    53→function deriveStatus(dep: Dependency): StatusBadgeType {
    54→  if (!dep.is_active) return "unknown";
    55→  return "operational";
    56→}
    57→
    58→const STATUS_CONFIG: Record<
    59→  StatusBadgeType,
    60→  { label: string; color: string; bg: string }
    61→> = {
    62→  operational: {
    63→    label: "Operational",
    64→    color: "#16A34A",
    65→    bg: "rgba(22,163,74,0.12)",
    66→  },
    67→  degraded: {
    68→    label: "Degraded",
    69→    color: "#D97706",
    70→    bg: "rgba(217,119,6,0.12)",
    71→  },
    72→  down: {
    73→    label: "Down",
    74→    color: "#DC2626",
    75→    bg: "rgba(220,38,38,0.12)",
    76→  },
    77→  unknown: {
    78→    label: "Unknown",
    79→    color: "#52525B",
    80→    bg: "rgba(82,82,91,0.12)",
    81→  },
    82→};
    83→
    84→function uptimeColor(pct: number | null): string {
    85→  if (pct === null) return "#52525B";
    86→  if (pct >= 99) return "#16A34A";
    87→  if (pct >= 95) return "#D97706";
    88→  return "#DC2626";
    89→}
    90→
    91→function formatUptime(pct: number | null): string {
    92→  if (pct === null) return "—";
    93→  return pct.toFixed(2) + "%";
    94→}

    96→// ── Component ────────────────────────────────────────────────────────────────
    97→
    98→export default function DependenciesPage() {

   100→
   101→  // TanStack Query hooks

   103→  const { data: billingPlan } = useBillingPlan();
   104→  const createMutation = useCreateDependency();
   105→  const updateMutation = useUpdateDependency();
   106→  const deleteMutation = useDeleteDependency();
   107→
   108→  const plan = (billingPlan?.plan || "free") as Plan;
   109→
   110→  // History state (fetched on-demand per dependency)

   112→
   113→  // Modal

   115→  const [formName, setFormName] = useState("");

   118→  const [formInterval, setFormInterval] = useState(60);
   119→  const [formRegions, setFormRegions] = useState<string[]>(["us-east"]);
   120→  const [formEndpointError, setFormEndpointError] = useState("");
   121→
   122→  // Actions
   123→  const [activeMenuId, setActiveMenuId] = useState<string | null>(null);

   125→  const menuRef = useRef<HTMLDivElement>(null);
   126→
   127→  // ── Derived ──────────────────────────────────────────────────────────────
   128→

   130→  const depLimit = planConfig.limits.dependencies;
   131→  const maxRegions = planConfig.limits.regions;
   132→  const minInterval = planConfig.limits.interval;

   134→  const isFreeOrStarter = plan === "free" || plan === "starter";

   138→    dependencies.forEach((dep) => {
   139→      if (!historyMap[dep.id]) {

   141→          setHistoryMap((prev) => ({ ...prev, [dep.id]: history }));

   143→          // silently ignore
   144→        });

   169→    setModalOpen(true);
   170→  };
   171→

   173→    // Validate HTTPS
   174→    if (
   175→      !formEndpoint.startsWith("https://") &&

   177→    ) {
   178→      setFormEndpointError("Endpoint must be a valid URL (https:// or http://)");
   179→      return;
   180→    }
   181→    if (!formName.trim() || !formEndpoint.trim()) {
   182→      setFormEndpointError("Name and endpoint URL are required.");
   183→      return;
   184→    }

   186→    try {
   187→      await createMutation.mutateAsync({
   188→        name: formName.trim(),
   189→        endpoint_url: formEndpoint.trim(),
   190→        expected_status_codes: [formExpectedStatus],
   191→        check_interval_seconds: formInterval,
   192→        regions: formRegions,
   193→      });
   194→      setModalOpen(false);
   195→    } catch {
   196→      setFormEndpointError("Failed to create dependency. Please try again.");
   197→    }

   199→
   200→  const handleToggle = async (dep: Dependency) => {
   201→    setActiveMenuId(null);
   202→    try {
   203→      await updateMutation.mutateAsync({
   204→        id: dep.id,
   205→        data: { is_active: !dep.is_active },
   206→      });
   207→    } catch {
   208→      // Silently ignore
   209→    }

   211→
   212→  const handleDelete = async (id: string) => {
   213→    setActiveMenuId(null);
   214→    try {
   215→      await deleteMutation.mutateAsync(id);
   216→    } catch {
   217→      // Silently ignore
   218→    }
   219→  };
   220→

   222→    setFormRegions((prev) => {
   223→      if (prev.includes(region)) {
   224→        if (prev.length > 1) return prev.filter((r) => r !== region);
   225→        return prev;
   226→      }
   227→      if (plan === "free" && prev.length >= 1) return prev;

   229→      return [...prev, region];
   230→    });
   231→  };
   232→
   233→  // ── Render ──────────────────────────────────────────────────────────────
   234→

   236→    <div className="min-h-screen bg-[#0A0A0F]">
   237→      <div className="max-w-6xl mx-auto px-6 py-8 space-y-6">

   239→        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   240→          <div>
   241→            <h1 className="text-2xl font-bold text-[#FAFAFA]">Dependencies</h1>
   242→            <p className="text-sm text-[#52525B] mt-1">
   243→              Monitor and manage your external service endpoints
   244→            </p>
   245→          </div>
   246→          <button
   247→            onClick={openModal}
   248→            className="bg-[#FAFAFA] text-[#0A0A0F] px-4 py-2.5 rounded-lg text-sm font-semibold hover:bg-white transition-colors inline-flex items-center gap-2 shrink-0"
   249→          >
   250→            <Plus className="w-4 h-4" />
   251→            Add dependency

   253→        </div>

   255→        {/* Upgrade Banner */}
   256→        {isFreeOrStarter && (
   257→          <UpgradeBanner
   258→            plan={plan}
   259→            usage={dependencies.length}
   260→            limit={depLimit}
   261→            resource="dependencies"
   262→            onUpgrade={() => {}}
   263→          />
   264→        )}
   265→
   266→        {/* Error */}

   268→          <div className="rounded-xl border border-[rgba(220,38,38,0.2)] bg-[rgba(220,38,38,0.08)] px-5 py-4 flex items-start gap-3">
   269→            <div className="w-2 h-2 rounded-full bg-[#DC2626] mt-1.5 shrink-0" />
   270→            <p className="text-sm text-[#FAFAFA]">Unable to load dependencies. Please try again.</p>
   271→            <button
   272→              onClick={() => refetch()}
   273→              className="text-xs font-medium text-[#0891B2] ml-auto shrink-0 hover:underline"
   274→            >
   275→              Retry
   276→            </button>
   277→          </div>
   278→        )}
   279→
   280→        {/* Loading Skeleton */}

   282→          <ConsoleCard>
   283→            <div className="px-5 py-3 grid grid-cols-[1.5fr,1fr,auto,auto,auto,auto] gap-4">
   284→              {Array.from({ length: 6 }).map((_, i) => (
   285→                <Skeleton
   286→                  key={i}
   287→                  className="h-3 w-12 rounded bg-[rgba(255,255,255,0.06)]"
   288→                />
   289→              ))}
   290→            </div>
   291→            {Array.from({ length: 5 }).map((_, rowIdx) => (
   292→              <div
   293→                key={rowIdx}
   294→                className="px-5 py-4 grid grid-cols-[1.5fr,1fr,auto,auto,auto,auto] gap-4 border-t border-[rgba(255,255,255,0.05)]"
   295→              >
   296→                {Array.from({ length: 6 }).map((_, i) => (
   297→                  <Skeleton
   298→                    key={i}
   299→                    className="h-4 w-full rounded bg-[rgba(255,255,255,0.06)]"
   300→                  />
   301→                ))}
   302→              </div>
   303→            ))}
   304→          </ConsoleCard>
   305→        )}
   306→
   307→        {/* Empty State */}

   309→          <EmptyState
   310→            icon={Layers}
   311→            title="No dependencies configured"
   312→            description="Add a dependency to start monitoring its uptime and reliability. Track response times, get alerted on outages, and correlate incidents automatically."
   313→            actionLabel="Add dependency"
   314→            onAction={openModal}
   315→          />
   316→        )}
   317→
   318→        {/* Dependencies Table */}
   319→        {!loading && dependencies.length > 0 && (
   320→          <ConsoleCard className="overflow-hidden">
   321→            {/* Table Header */}
   322→            <div className="px-5 py-3 grid grid-cols-[1.5fr,1fr,auto,auto,auto,auto] gap-4 text-[11px] font-semibold uppercase tracking-wider text-[#52525B] bg-[rgba(255,255,255,0.02)]">
   323→              <div>Name</div>
   324→              <div>Endpoint</div>
   325→              <div>Status</div>
   326→              <div>Uptime</div>
   327→              <div>Last check</div>
   328→              <div className="w-8" />
   329→            </div>
   330→
   331→            {/* Rows */}
   332→            {dependencies.map((dep, idx) => {
   333→              const status = deriveStatus(dep);
   334→              const statusCfg = STATUS_CONFIG[status];
   335→              const history = historyMap[dep.id] || null;
   336→              const uptime = history ? history.uptime_percentage : null;
   337→
   338→              return (
   339→                <div
   340→                  key={dep.id}
   341→                  className="px-5 py-3.5 grid grid-cols-[1.5fr,1fr,auto,auto,auto,auto] gap-4 border-t border-[rgba(255,255,255,0.05)] hover:bg-[rgba(255,255,255,0.02)] transition-colors items-center"
   342→                >
   343→                  {/* Name */}
   344→                  <div className="min-w-0">
   345→                    <div className="text-sm font-medium text-[#FAFAFA] truncate">
   346→                      {dep.name}
   347→                    </div>
   348→                    <div className="text-xs text-[#52525B] font-mono truncate mt-0.5">
   349→                      {dep.regions.length > 0
   350→                        ? dep.regions.join(", ")
   351→                        : "—"}
   352→                    </div>
   353→                  </div>
   354→
   355→                  {/* Endpoint */}
   356→                  <div className="min-w-0">
   357→                    <div className="text-xs text-[#A1A1AA] font-mono truncate">
   358→                      {dep.endpoint_url}
   359→                    </div>
   360→                  </div>
   361→
   362→                  {/* Status */}
   363→                  <div className="flex items-center gap-2">
   364→                    <StatusDot status={status} pulse={status === "down"} />
   365→                    <span
   366→                      className="text-xs font-medium px-2 py-0.5 rounded-full"
   367→                      style={{
   368→                        color: statusCfg.color,
   369→                        backgroundColor: statusCfg.bg,
   370→                      }}
   371→                    >
   372→                      {statusCfg.label}
   373→                    </span>
   374→                  </div>
   375→
   376→                  {/* Uptime */}
   377→                  <div
   378→                    className="font-mono text-sm"
   379→                    style={{ color: uptimeColor(uptime) }}
   380→                  >
   381→                    {formatUptime(uptime)}
   382→                  </div>
   383→
   384→                  {/* Last check */}
   385→                  <div className="text-xs text-[#52525B] font-mono whitespace-nowrap">
   386→                    {dep.updated_at
   387→                      ? formatDistanceToNow(new Date(dep.updated_at), {
   388→                          addSuffix: true,
   389→                        })
   390→                      : "—"}
   391→                  </div>
   392→
   393→                  {/* Actions */}
   394→                  <div className="relative w-8 flex justify-center" ref={idx === dependencies.findIndex(d => d.id === activeMenuId) ? menuRef : undefined}>
   395→                    <button
   396→                      onClick={() =>
   397→                        setActiveMenuId(
   398→                          activeMenuId === dep.id ? null : dep.id
   399→                        )
   400→                      }
   401→                      className="p-1 rounded-md hover:bg-[rgba(255,255,255,0.08)] transition-colors text-[#52525B] hover:text-[#A1A1AA]"
   402→                    >
   403→                      <MoreVertical className="w-4 h-4" />
   404→                    </button>
   405→
   406→                    {/* Dropdown */}
   407→                    {activeMenuId === dep.id && (
   408→                      <div className="absolute right-0 top-full mt-1 z-20 bg-[#1A1A20] border border-[rgba(255,255,255,0.08)] rounded-lg shadow-console-dropdown py-1 min-w-[160px]">
   409→                        <button
   410→                          className="w-full flex items-center gap-2.5 px-3 py-2 text-sm text-[#A1A1AA] hover:text-[#FAFAFA] hover:bg-[rgba(255,255,255,0.06)] transition-colors"
   411→                          onClick={() => setActiveMenuId(null)}
   412→                        >
   413→                          <Edit3 className="w-3.5 h-3.5" />
   414→                          Edit
   415→                        </button>
   416→                        <button
   417→                          className={cn(
   418→                            "w-full flex items-center gap-2.5 px-3 py-2 text-sm text-[#A1A1AA] hover:text-[#FAFAFA] hover:bg-[rgba(255,255,255,0.06)] transition-colors",
   419→                            updateMutation.isPending && "opacity-50 pointer-events-none"
   420→                          )}
   421→                          onClick={() => handleToggle(dep)}
   422→                        >
   423→                          {dep.is_active ? (
   424→                            <>
   425→                              <Pause className="w-3.5 h-3.5" />
   426→                              Pause
   427→                            </>
   428→                          ) : (
   429→                            <>
   430→                              <Play className="w-3.5 h-3.5" />
   431→                              Resume
   432→                            </>
   433→                          )}
   434→                        </button>
   435→                        <div className="my-1 border-t border-[rgba(255,255,255,0.06)]" />
   436→                        <button
   437→                          className={cn(
   438→                            "w-full flex items-center gap-2.5 px-3 py-2 text-sm text-[#DC2626] hover:bg-[rgba(220,38,38,0.08)] transition-colors",
   439→                            deleteMutation.isPending && "opacity-50 pointer-events-none"
   440→                          )}
   441→                          onClick={() => handleDelete(dep.id)}
   442→                        >
   443→                          <Trash2 className="w-3.5 h-3.5" />
   444→                          Delete
   445→                        </button>
   446→                      </div>
   447→                    )}
   448→                  </div>
   449→                </div>
   450→              );
   451→            })}
   452→          </ConsoleCard>
   453→        )}
   454→      </div>
   455→
   456→      {/* ── Add Dependency Modal ──────────────────────────────────────────────── */}
   457→      {modalOpen && (

   459→          className="fixed inset-0 bg-black/70 backdrop-blur-sm z-50 flex items-center justify-center p-4"
   460→          onClick={(e) => {
   461→            if (e.target === e.currentTarget) setModalOpen(false);
   462→          }}
   463→        >
   464→          <div className="bg-[#1A1A20] rounded-2xl border border-[rgba(255,255,255,0.08)] max-w-lg w-full p-6 animate-[fadeIn_200ms_ease-out]">
   465→            {/* Modal Header */}
   466→            <div className="flex items-center justify-between mb-6">
   467→              <h2 className="text-lg font-semibold text-[#FAFAFA]">
   468→                Add dependency
   469→              </h2>
   470→              <button
   471→                onClick={() => setModalOpen(false)}
   472→                className="p-1.5 rounded-lg hover:bg-[rgba(255,255,255,0.08)] transition-colors text-[#52525B] hover:text-[#A1A1AA]"
   473→              >
   474→                <X className="w-4 h-4" />
   475→              </button>
   476→            </div>
   477→
   478→            {/* Form */}
   479→            <div className="space-y-4">
   480→              {/* Name */}
   481→              <div>
   482→                <label className="text-sm font-medium text-[#A1A1AA] mb-1.5 block">
   483→                  Name
   484→                </label>
   485→                <input
   486→                  type="text"
   487→                  value={formName}
   488→                  onChange={(e) => setFormName(e.target.value)}
   489→                  placeholder="Stripe API"
   490→                  className="w-full bg-[#1C1C22] border border-[rgba(255,255,255,0.08)] text-[#FAFAFA] text-sm rounded-lg px-3.5 py-2.5 placeholder:text-[#52525B] focus:outline-none focus:border-[rgba(8,145,178,0.5)] transition-colors"
   491→                />
   492→              </div>
   493→
   494→              {/* Endpoint URL */}
   495→              <div>
   496→                <label className="text-sm font-medium text-[#A1A1AA] mb-1.5 block">
   497→                  Endpoint URL
   498→                </label>
   499→                <input
   500→                  type="text"
   501→                  value={formEndpoint}
   502→                  onChange={(e) => {
   503→                    setFormEndpoint(e.target.value);
   504→                    setFormEndpointError("");
   505→                  }}
   506→                  placeholder="https://api.stripe.com/v1/charges"
   507→                  className={cn(
   508→                    "w-full bg-[#1C1C22] border text-[#FAFAFA] text-sm rounded-lg px-3.5 py-2.5 placeholder:text-[#52525B] focus:outline-none focus:border-[rgba(8,145,178,0.5)] transition-colors font-mono",
   509→                    formEndpointError
   510→                      ? "border-[rgba(220,38,38,0.5)]"
   511→                      : "border-[rgba(255,255,255,0.08)]"
   512→                  )}
   513→                />
   514→                {formEndpointError && (
   515→                  <p className="text-xs text-[#DC2626] mt-1.5">
   516→                    {formEndpointError}
   517→                  </p>

   519→              </div>

   521→              {/* Expected status + Check interval */}

   525→                    Expected status

   544→                  >
