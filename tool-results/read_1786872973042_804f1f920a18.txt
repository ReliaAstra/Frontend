    37→function formatDuration(seconds: number): string {
    38→  if (seconds < 60) return `${seconds}s`;
    39→  if (seconds < 3600) return `${Math.floor(seconds / 60)}m ${seconds % 60}s`;
    40→  const h = Math.floor(seconds / 3600);
    41→  const m = Math.floor((seconds % 3600) / 60);
    42→  return `${h}h ${m}m`;
    43→}
