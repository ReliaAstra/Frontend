    75→function normalizeStatus(raw: string): 'operational' | 'degraded' | 'down' | 'unknown' {
    76→  const s = raw.toLowerCase();
    77→  if (s === 'up' || s === 'operational') return 'operational';
    78→  if (s === 'degraded' || s === 'degraded_performance' || s === 'partial_outage') return 'degraded';
    79→  if (s === 'down' || s === 'major_outage') return 'down';
    80→  return 'unknown';

    88→// ── Animated Counter Hook ──────────────────────────────────────────────────────
    89→
    90→function useAnimatedCounter(target: number, duration = 600) {
    91→  const [value, setValue] = useState(0);
    92→
    93→  useEffect(() => {
    94→    if (prefersReducedMotion) {
    95→      setValue(target);
    96→      return;
    97→    }
    98→    let start = 0;
    99→    let raf: number;
   100→    const step = (timestamp: number) => {
   101→      if (!start) start = timestamp;
   102→      const progress = Math.min((timestamp - start) / duration, 1);
   103→      const eased = 1 - Math.pow(1 - progress, 3);
   104→      setValue(Math.round(eased * target));
   105→      if (progress < 1) {
   106→        raf = requestAnimationFrame(step);
   107→      }
   108→    };
   109→    raf = requestAnimationFrame(step);
   110→    return () => cancelAnimationFrame(raf);
   111→  }, [target, duration]);
   112→
   113→  return value;
   114→}

   320→              metricsRes.status === 'fulfilled' ? metricsRes.value : null;

   324→            // Build sparkline from timeline-like data if available, else flat
