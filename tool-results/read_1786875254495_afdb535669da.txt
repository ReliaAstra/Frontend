    82→

    84→  const { currentOrg, isLoading: authLoading } = useAuth();
    85→
    86→  /* ── State ──────────────────────────────────────────────────────────────── */
    87→  const [summary, setSummary] = useState<DashboardSummaryResponse | null>(null);
    88→  const [incidents, setIncidents] = useState<Incident[]>([]);
    89→  const [clients, setClients] = useState<Client[]>([]);

    92→    total_degradation_pct: 0,

   103→  const isAgency = currentOrg?.has_agency_mode === true;
   104→  const isFreePlan = plan === "free";
   105→  const planConfig = getPlanConfig(plan);
   106→  const depLimit = planConfig.limits.dependencies;
   107→  const atDepLimit =

   109→    isFinite(depLimit) &&
   110→    summary.active_dependencies_count >= depLimit;

   112→    (i) => i.evidence_report_id !== null
   113→  ).length;

   120→        : "#DC2626";

   122→  /* ── Data fetching ────────────────────────────────────────────────────── */
   123→  const fetchDashboard = useCallback(
   124→    async (isRefresh = false) => {
   125→      if (isRefresh) setRefreshing(true);
   126→      else setLoading(true);
   127→      setError(null);
   128→
   129→      try {
   130→        // Core fetches — failure triggers the error banner
   131→        const [summaryData, incidentsData] = await Promise.all([
   132→          dashboardService.getSummary(),
   133→          incidentService.list("open"),
   134→        ]);
   135→        setSummary(summaryData);
   136→        setIncidents(incidentsData);
   137→        setLastUpdated(new Date());

   139→        // Non-critical fetches — degrade gracefully
   140→        try {
   141→          const clientRes = await clientService.list({ per_page: 5 });
   142→          setClients(
   143→            Array.isArray(clientRes)
   144→              ? clientRes
   145→              : clientRes?.items ?? []
   146→          );
   147→        } catch {
   148→          setClients([]);
   149→        }
   150→
   151→        try {
   152→          const planRes = await billingService.getPlan();
   153→          setPlan(planRes.plan);
   154→        } catch {
   155→          setPlan("free");
   156→        }
   157→
   158→        try {
   159→          const checks = await dashboardService.getRecentChecks(5);
   160→          setRecentChecks(checks);
   161→        } catch {
   162→          setRecentChecks([]);
   163→        }
   164→
   165→        try {
   166→          const sla = await dashboardService.getSlaDegradation(30);
   167→          setSlaDegradation(sla);
   168→        } catch {
   169→          setSlaDegradation({
   170→            total_degradation_pct: 0,
   171→            affected_services: 0,
   172→            period: "30d",
   173→          });
   174→        }
   175→      } catch {
   176→        setError(
   177→          "Unable to load dashboard data."
   178→        );
   179→      } finally {
   180→        setLoading(false);
   181→        setRefreshing(false);
   182→      }
   183→    },
   184→    []
   185→  );
   186→
   187→  useEffect(() => {
   188→    if (!authLoading) fetchDashboard();
   189→  }, [authLoading, fetchDashboard]);
   190→
   191→  /* ── Auth loading shell ────────────────────────────────────────────────── */
   192→  if (authLoading) {
   193→    return (
   194→      <div className="space-y-6 animate-[fadeIn_200ms_ease-out]">
   195→        {/* Header skeleton */}
   196→        <div className="flex items-start justify-between">
   197→          <div>
   198→            <Skeleton className="h-8 w-56 mb-2" />
   199→            <Skeleton className="h-4 w-44" />
   200→          </div>
   201→          <Skeleton className="h-10 w-10 rounded-lg" />
   202→        </div>
   203→
   204→        {/* KPI skeletons */}
   205→        <div
   206→          className={cn(
   207→            "grid grid-cols-2 gap-4",
   208→            isAgency ? "lg:grid-cols-5" : "lg:grid-cols-4"
   209→          )}
   210→        >
   211→          {Array.from({ length: isAgency ? 5 : 4 }).map((_, i) => (
   212→            <KpiCardSkeleton key={i} />
   213→          ))}
   214→        </div>
   215→
   216→        {/* Two-column skeletons */}
   217→        <div className="grid grid-cols-1 lg:grid-cols-[1fr_380px] gap-4">
   218→          <SectionSkeleton />
   219→          <div className="space-y-4">
   220→            <SectionSkeleton />
   221→            <SectionSkeleton />
   222→          </div>
   223→        </div>
   224→      </div>
   225→    );
   226→  }
   227→
   228→  /* ══════════════════════════════════════════════════════════════════════════ */
   229→  return (
   230→    <div className="space-y-6 animate-[fadeIn_200ms_ease-out]">
   231→      {/* ────────────────────────── Header ─────────────────────────────── */}

   233→        <div>
   234→          <h1 className="text-2xl font-bold text-[#FAFAFA]">
   235→            Operations overview
   236→          </h1>
   237→          <p className="text-sm text-[#52525B] mt-1">
   238→            {currentOrg?.name || "Organization"}&nbsp;·&nbsp;Last updated{" "}
   239→            {lastUpdated
   240→              ? formatDistanceToNow(lastUpdated, { addSuffix: true })
   241→              : "just now"}
   242→          </p>
   243→        </div>
   244→        <button
   245→          onClick={() => fetchDashboard(true)}
   246→          disabled={refreshing || loading}
   247→          className="p-2.5 rounded-lg bg-[#131318] border border-[rgba(255,255,255,0.08)] text-[#A1A1AA] hover:text-[#FAFAFA] hover:border-[rgba(255,255,255,0.15)] transition-all disabled:opacity-50"
   248→          aria-label="Refresh dashboard"
   249→        >
   250→          <RefreshCw
   251→            className={cn("w-4 h-4", refreshing && "animate-spin")}
   252→          />
   253→        </button>
   254→      </div>
   255→
   256→      {/* ────────────────────── Upgrade Banner ────────────────────────── */}
   257→      {(plan === "free" || plan === "starter") && summary && (
   258→        <UpgradeBanner
   259→          plan={plan}
   260→          usage={summary.active_dependencies_count}
   261→          limit={depLimit}
   262→          resource="dependencies"

   265→
   266→      {/* ────────────────────── Error Banner ───────────────────────────── */}

   269→          <AlertCircle className="w-5 h-5 text-[#DC2626] mt-0.5 shrink-0" />
   270→          <div className="flex-1">
   271→            <p className="text-sm font-semibold text-[#FAFAFA]">{error}</p>
   272→            <p className="text-xs text-[#A1A1AA] mt-0.5">
   273→              Check that the backend is accessible and your session is valid.
   274→            </p>
   275→          </div>
   276→          <button
   277→            onClick={() => fetchDashboard()}
   278→            className="text-xs font-semibold text-[#DC2626] hover:text-[#FAFAFA] transition-colors"
   279→          >
   280→            Retry
   281→          </button>
   282→        </div>
   283→      )}
   284→
   285→      {/* ────────────────── Global Stats Row ──────────────────────────── */}

   290→            isAgency ? "lg:grid-cols-5" : "lg:grid-cols-4"

   295→          ))}
   296→        </div>
   297→      ) : summary ? (
   298→        <div
   299→          className={cn(
   300→            "grid grid-cols-2 gap-4",
   301→            isAgency ? "lg:grid-cols-5" : "lg:grid-cols-4"

   303→        >

   305→          {isAgency && (
   306→            <div className="bg-[#131318] rounded-xl border border-[rgba(255,255,255,0.08)] p-5">
   307→              <div className="flex items-center gap-2">
   308→                <Users className="w-4 h-4 text-[#52525B]" />
   309→                <span className="text-[11px] uppercase tracking-wider text-[#52525B]">
   310→                  Clients
   311→                </span>
   312→              </div>
   313→              <MetricValue value={clients.length} className="mt-2" />
   314→              <p className="text-xs text-[#52525B] mt-1">
   315→                {clients.length} managed
   316→              </p>
   317→            </div>
   318→          )}
   319→
   320→          {/* ── Dependencies ──────────────────────────────────────────── */}
   321→          <div className="bg-[#131318] rounded-xl border border-[rgba(255,255,255,0.08)] p-5">
   322→            <div className="flex items-center gap-2">
   323→              <Layers className="w-4 h-4 text-[#52525B]" />
   324→              <span className="text-[11px] uppercase tracking-wider text-[#52525B]">
   325→                Dependencies
   326→              </span>
   327→              {atDepLimit && (
   328→                <span className="text-[10px] font-semibold text-[#DC2626] bg-[rgba(220,38,38,0.12)] px-2 py-0.5 rounded-full">
   329→                  At limit
   330→                </span>
   331→              )}
   332→            </div>
   333→            <MetricValue
   334→              value={summary.active_dependencies_count}
   335→              className="mt-2"
   336→              color={atDepLimit ? "#DC2626" : undefined}
   337→            />
   338→            <p className="text-xs text-[#52525B] mt-1">
   339→              of {depLimit === Infinity ? "\u221E" : depLimit} allowed
   340→            </p>
   341→          </div>
   342→
   343→          {/* ── Open Incidents ────────────────────────────────────────── */}
   344→          <div
   345→            className={cn(
   346→              "bg-[#131318] rounded-xl border p-5 transition-colors",
   347→              summary.open_incidents_count > 0
   348→                ? "border-[rgba(220,38,38,0.2)]"
   349→                : "border-[rgba(255,255,255,0.08)]"
   350→            )}
   351→          >
   352→            <div className="flex items-center gap-2">
   353→              <AlertTriangle
   354→                className={cn(
   355→                  "w-4 h-4",
   356→                  summary.open_incidents_count > 0
   357→                    ? "text-[#DC2626] animate-pulse"
   358→                    : "text-[#52525B]"
   359→                )}
   360→              />
   361→              <span className="text-[11px] uppercase tracking-wider text-[#52525B]">
   362→                Open Incidents
   363→              </span>
   364→            </div>
   365→            <MetricValue
   366→              value={summary.open_incidents_count}
   367→              className="mt-2"
   368→            />
   369→            <p className="text-xs text-[#52525B] mt-1">
   370→              {summary.open_incidents_count > 0
   371→                ? "requiring attention"
   372→                : "all clear"}
   373→            </p>
   374→          </div>
   375→
   376→          {/* ── Evidence ───────────────────────────────────────────────── */}
   377→          {isFreePlan ? (
   378→            <div className="bg-[#131318] rounded-xl border border-[rgba(255,255,255,0.08)] p-5">
   379→              <div className="flex items-center gap-2">
   380→                <FileText className="w-4 h-4 text-[#52525B]" />
   381→                <span className="text-[11px] uppercase tracking-wider text-[#52525B]">
   382→                  Evidence
   383→                </span>
   384→                <Lock className="w-3 h-3 text-[#52525B]" />
   385→              </div>
   386→              <MetricValue value="\u2014" className="mt-2" />
   387→              <p className="text-xs text-[#52525B] mt-1">
   388→                Upgrade to generate
   389→              </p>
   390→            </div>
   391→          ) : (
   392→            <div className="bg-[#131318] rounded-xl border border-[rgba(255,255,255,0.08)] p-5">
   393→              <div className="flex items-center gap-2">
   394→                <FileText className="w-4 h-4 text-[#52525B]" />
   395→                <span className="text-[11px] uppercase tracking-wider text-[#52525B]">
   396→                  Evidence
   397→                </span>
   398→              </div>
   399→              <MetricValue value={evidenceCount} className="mt-2" />
   400→              <p className="text-xs text-[#52525B] mt-1">reports generated</p>
   401→            </div>
   402→          )}
   403→
   404→          {/* ── SLA ───────────────────────────────────────────────────── */}
   405→          <div className="bg-[#131318] rounded-xl border border-[rgba(255,255,255,0.08)] p-5">
   406→            <div className="flex items-center gap-2">
   407→              <ShieldCheck className="w-4 h-4 text-[#52525B]" />
   408→              <span className="text-[11px] uppercase tracking-wider text-[#52525B]">
   409→                SLA
   410→              </span>
   411→            </div>
   412→            <MetricValue
   413→              value={`${slaDegradation.total_degradation_pct.toFixed(2)}%`}
   414→              className="mt-2"
   415→              color={slaColor}
   416→            />
   417→            <p className="text-xs text-[#52525B] mt-1">degradation (30d)</p>
   418→          </div>
   419→        </div>

   421→
   422→      {/* ────────────────── Active Incidents ─────────────────────────── */}
   423→      <div>
   424→        <div className="flex items-center justify-between mb-4">
   425→          <h2 className="text-lg font-semibold text-[#FAFAFA]">
   426→            Active incidents
   427→          </h2>
   428→          <Link
   429→            href="/incidents"
   430→            className="text-sm text-[#A1A1AA] hover:text-[#FAFAFA] transition-colors flex items-center gap-1"

   432→            View all

   434→          </Link>
   435→        </div>
   436→
   437→        {loading ? (
   438→          <ConsoleCard>
   439→            <ConsoleCardBody>
   440→              <div className="space-y-3">
   441→                {Array.from({ length: 3 }).map((_, i) => (
   442→                  <Skeleton key={i} className="h-14 w-full" />

   445→            </ConsoleCardBody>
   446→          </ConsoleCard>
   447→        ) : incidents.length > 0 ? (
   448→          <ConsoleCard>
   449→            {/* Table header */}
   450→            <div
   451→              className={cn(
   452→                "px-5 py-3 grid gap-4 text-[11px] font-semibold uppercase tracking-wider text-[#52525B] bg-[rgba(255,255,255,0.02)] border-b border-[rgba(255,255,255,0.05)]",
   453→                INCIDENT_GRID_COLS

   456→              <span>Severity</span>
   457→              <span>Title</span>
   458→              <span>Dependency</span>
   459→              <span>Opened</span>
