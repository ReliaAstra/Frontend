     1→'use client';
     2→
     3→import { useState, useEffect, useCallback, useMemo } from 'react';
     4→import { motion } from 'framer-motion';
     5→import {
     6→  Bell, ExternalLink, CheckCircle2, AlertTriangle, Shield,
     7→  TrendingUp, RefreshCw, AlertCircle, ArrowRight,
     8→} from 'lucide-react';
     9→import { Button } from '@/components/ui/button';
    10→import { Input } from '@/components/ui/input';
    11→import { Skeleton } from '@/components/ui/skeleton';
    12→import { cn } from '@/lib/utils';
    13→import { toast } from 'sonner';
    14→import {
    15→  vendorService,
    16→  type VendorDetailResponse,
    17→  type VendorHistoryResponse,

    19→  type VendorIncidentsResponse,
    20→  type VendorIncident,
    21→} from '@/services/vendorService';
    22→import { formatDistanceToNow, format } from 'date-fns';

    24→
    25→const statusConfig: Record<string, { label: string; dotColor: string; textColor: string; bg: string; borderColor: string }> = {
    26→  up: { label: 'Operational', dotColor: 'bg-emerald-500', textColor: 'text-emerald-700', bg: 'bg-emerald-50', borderColor: 'border-emerald-200' },
    27→  operational: { label: 'Operational', dotColor: 'bg-emerald-500', textColor: 'text-emerald-700', bg: 'bg-emerald-50', borderColor: 'border-emerald-200' },
    28→  degraded: { label: 'Degraded', dotColor: 'bg-amber-500', textColor: 'text-amber-700', bg: 'bg-amber-50', borderColor: 'border-amber-200' },
    29→  degraded_performance: { label: 'Degraded', dotColor: 'bg-amber-500', textColor: 'text-amber-700', bg: 'bg-amber-50', borderColor: 'border-amber-200' },
    30→  down: { label: 'Down', dotColor: 'bg-red-500', textColor: 'text-red-700', bg: 'bg-red-50', borderColor: 'border-red-200' },

    32→  major_outage: { label: 'Major Outage', dotColor: 'bg-red-500', textColor: 'text-red-700', bg: 'bg-red-50', borderColor: 'border-red-200' },

    34→};
    35→
    36→const severityConfig: Record<string, { label: string; color: string }> = {
    37→  critical: { label: 'Critical', color: 'text-red-700 bg-red-50 border-red-200' },
    38→  major: { label: 'Major', color: 'text-orange-700 bg-orange-50 border-orange-200' },
    39→  high: { label: 'High', color: 'text-orange-700 bg-orange-50 border-orange-200' },
    40→  minor: { label: 'Minor', color: 'text-amber-700 bg-amber-50 border-amber-200' },
    41→  medium: { label: 'Medium', color: 'text-amber-700 bg-amber-50 border-amber-200' },
    42→  low: { label: 'Low', color: 'text-blue-700 bg-blue-50 border-blue-200' },
    43→};
    44→
    45→interface Props {
    46→  vendorSlug: string;
    47→}
    48→
    49→export function TrackVendorContent({ vendorSlug }: Props) {
    50→  const [email, setEmail] = useState('');
    51→  const [subscribing, setSubscribing] = useState(false);
    52→
    53→  const [detail, setDetail] = useState<VendorDetailResponse | null>(null);
    54→  const [history, setHistory] = useState<VendorHistoryResponse | null>(null);
    55→  const [metrics, setMetrics] = useState<VendorMetricsResponse | null>(null);
    56→  const [incidents, setIncidents] = useState<VendorIncidentsResponse | null>(null);
    57→  const [loading, setLoading] = useState(true);
    58→  const [fetchError, setFetchError] = useState<string | null>(null);
    59→  const [lastFetch, setLastFetch] = useState<string | null>(null);
    60→  const [refreshing, setRefreshing] = useState(false);
    61→
    62→  const vendorLabel = detail?.display_name || vendorSlug.split('-').map(w => w.charAt(0).toUpperCase() + w.slice(1)).join(' ');
    63→  const vendorName = detail?.vendor_name || vendorSlug;
    64→
    65→  const fetchData = useCallback(async (isRefresh = false) => {
    66→    if (isRefresh) setRefreshing(true);
    67→    else setLoading(true);
    68→    setFetchError(null);
    69→    try {
    70→      const [d, h, m, i] = await Promise.all([
    71→        vendorService.getVendorDetail(vendorSlug),
    72→        vendorService.getVendorHistory(vendorSlug),
    73→        vendorService.getVendorMetrics(vendorSlug),
    74→        vendorService.getVendorIncidents(vendorSlug, 50),
    75→      ]);
    76→      setDetail(d);
    77→      setHistory(h);
    78→      setMetrics(m);
    79→      setIncidents(i);
    80→      setLastFetch(new Date().toISOString());
    81→    } catch {
    82→      setFetchError('Unable to load vendor intelligence data. The measurement service may be unavailable.');
    83→    } finally {
    84→      setLoading(false);
    85→      setRefreshing(false);
    86→    }
    87→  }, [vendorSlug]);
    88→
    89→  useEffect(() => {
    90→    fetchData();
    91→    // Auto-refresh every 60s
    92→    const interval = setInterval(() => fetchData(true), 60000);
    93→    return () => clearInterval(interval);
    94→  }, [fetchData]);
    95→
    96→  // Process metrics for chart
    97→  const chartData = useMemo(() => {
    98→    if (!metrics?.metrics) return [];
    99→    return Object.entries(metrics.metrics)
   100→      .sort(([a], [b]) => a.localeCompare(b))
   101→      .map(([, w]) => ({
   102→        hour: w.window,
   103→        window: w.window,
   104→        latency: Math.round(w.avg_latency_ms),
   105→        p95: w.p95_latency_ms != null ? Math.round(w.p95_latency_ms) : 0,
   106→        uptime: w.uptime_percentage,
   107→        observations: w.total_observations,
   108→      }));
   109→  }, [metrics]);
   110→
   111→  const maxLatency = chartData.length > 0 ? Math.max(...chartData.map(d => d.latency), ...chartData.map(d => d.p95), 1) : 1;
   112→  const isZeroLatency = maxLatency === 0;
   113→
   114→  const overallStatus = detail?.recent_status || 'unknown';
   115→  const statusStyle = statusConfig[overallStatus] || statusConfig.unknown;
   116→
   117→  const uptimeValue = history?.uptime_percentage_24h ?? null;
   118→  const avgLatency = history?.avg_latency_ms_24h ?? null;
   119→  const totalChecks = history?.recent_checks_count ?? null;
   120→  const latestP95 = chartData.length > 0 ? chartData[chartData.length - 1].p95 : null;
   121→  const latestObservations = chartData.length > 0 ? chartData[chartData.length - 1].observations : null;
   122→
   123→  const isStale = detail?.last_check_at && (Date.now() - new Date(detail.last_check_at).getTime()) > 300000; // 5 min
   124→
   125→  const handleSubscribe = async () => {
   126→    if (!email.includes('@')) {
   127→      toast.error('Please enter a valid email address.');
   128→      return;
   129→    }
   130→    setSubscribing(true);
   131→    try {
   132→      const res = await fetch('/api/newsletter/subscribe', {
   133→        method: 'POST',
   134→        headers: { 'Content-Type': 'application/json' },
   135→        body: JSON.stringify({ email, vendor: vendorSlug }),
   136→      });
   137→      if (res.ok) {
   138→        toast.success(`Subscribed to ${vendorLabel} alerts.`);
   139→        setEmail('');
   140→      } else {
   141→        toast.error('Subscription failed. Please try again.');
   142→      }
   143→    } catch {
   144→      toast.error('Network error.');
   145→    } finally {
   146→      setSubscribing(false);
   147→    }
   148→  };
   149→
   150→  const formatDuration = (seconds: number | null) => {
   151→    if (seconds === null) return 'Ongoing';
   152→    if (seconds < 60) return `${Math.round(seconds)}s`;
   153→    if (seconds < 3600) return `${Math.floor(seconds / 60)}m ${Math.round(seconds % 60)}s`;
   154→    const h = Math.floor(seconds / 3600);
   155→    const m = Math.floor((seconds % 3600) / 60);
   156→    return `${h}h ${m}m`;
   157→  };
   158→
   159→  // --- RENDER ---
   160→
   161→  // Loading shell
   162→  if (loading) {
   163→    return (
   164→      <div className="max-w-6xl mx-auto px-4 md:px-8 py-12">
   165→        <Skeleton className="h-6 w-48 mb-4" />
   166→        <Skeleton className="h-10 w-80 mb-2" />
   167→        <Skeleton className="h-5 w-96 mb-8" />
   168→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-3 mb-8">
   169→          {Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-24 rounded-lg" />)}
   170→        </div>
   171→        <Skeleton className="h-64 rounded-lg mb-8" />
   172→        <Skeleton className="h-48 rounded-lg" />
   173→      </div>
   174→    );
   175→  }
   176→
   177→  // Error state
   178→  if (fetchError) {
   179→    return (
   180→      <div className="max-w-6xl mx-auto px-4 md:px-8 py-12">
   181→        <Link href="/track" className="text-sm text-[#0891B2] hover:text-[#0E7490] transition-colors mb-6 inline-block">
   182→          &larr; All Vendors
   183→        </Link>
   184→        <div className="rounded-lg border border-[#E4E4E7] bg-white p-12 text-center mt-4">
   185→          <AlertCircle className="h-8 w-8 text-[#A1A1AA] mx-auto mb-3" />
   186→          <p className="text-sm font-medium text-[#09090B]">{fetchError}</p>
   187→          <button
   188→            onClick={() => fetchData()}
   189→            className="mt-4 text-xs font-medium text-[#0891B2] hover:text-[#0E7490] transition-colors"
   190→          >
   191→            Retry
   192→          </button>
   193→        </div>
   194→      </div>
   195→    );
   196→  }
   197→
   198→  return (
   199→    <div className="max-w-6xl mx-auto px-4 md:px-8">
   200→      {/* ── HEADER ── */}
   201→      <section className="pt-10 pb-8">
   202→        <motion.div initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.4 }}>
   203→          {/* Breadcrumb */}
   204→          <div className="flex items-center gap-2 mb-6">
   205→            <Link href="/track" className="text-[13px] text-[#A1A1AA] hover:text-[#0891B2] transition-colors">
   206→              Vendors
   207→            </Link>
   208→            <span className="text-[#E4E4E7]">/</span>
   209→            <span className="text-[13px] text-[#52525B] font-medium">{vendorLabel}</span>
   210→          </div>
   211→
   212→          <div className="flex flex-col md:flex-row md:items-start md:justify-between gap-6">
   213→            <div>
   214→              {/* Vendor name + status */}
   215→              <div className="flex items-center gap-4 mb-3">
   216→                <h1 className="text-[28px] font-semibold text-[#09090B] tracking-[-0.02em]">
   217→                  {vendorLabel}
   218→                </h1>
   219→                <span className={cn(
   220→                  'inline-flex items-center gap-1.5 text-[11px] font-semibold uppercase tracking-[0.05em] px-2.5 py-1 rounded-md border',
   221→                  statusStyle.bg, statusStyle.textColor, statusStyle.borderColor
   222→                )}>
   223→                  <span className={cn('h-1.5 w-1.5 rounded-full', statusStyle.dotColor,
   224→                    (overallStatus === 'degraded' || overallStatus === 'degraded_performance' || overallStatus === 'partial_outage') && 'animate-pulse'
   225→                  )} />
   226→                  {statusStyle.label}
   227→                </span>
   228→              </div>
   229→
   230→              <p className="text-sm text-[#52525B] max-w-xl leading-relaxed">
   231→                {detail?.category ? `${detail.category} · ` : ''}Vendor intelligence measured continuously by Reliastra.
   232→                Data collected independently from {vendorLabel}&apos;s own status reporting.
   233→              </p>
   234→            </div>
   235→
   236→            {/* Refresh */}
   237→            <div className="flex items-center gap-3 shrink-0">
   238→              {lastFetch && (
   239→                <span className="text-[11px] font-mono text-[#A1A1AA]">
   240→                  {format(new Date(lastFetch), 'HH:mm:ss')} UTC
   241→                </span>
   242→              )}
   243→              <button
   244→                onClick={() => fetchData(true)}
   245→                disabled={refreshing}
   246→                className="p-2 rounded-md border border-[#E4E4E7] hover:bg-[#F8F9FA] transition-colors text-[#52525B] disabled:opacity-50"
   247→                aria-label="Refresh data"
   248→              >
   249→                <RefreshCw className={cn('w-3.5 h-3.5', refreshing && 'animate-spin')} />
   250→              </button>
   251→            </div>
   252→          </div>
   253→
   254→          {/* Live measurement bar */}
   255→          <div className="mt-5 px-4 py-3 rounded-lg bg-[#F8F9FA] border border-[#F0F0F0]">
   256→            <div className="flex flex-wrap items-center gap-x-6 gap-y-2 text-[12px]">
   257→              <div className="flex items-center gap-2">
   258→                <span className={cn(
   259→                  'h-1.5 w-1.5 rounded-full',
   260→                  isStale ? 'bg-amber-500 animate-pulse' : 'bg-emerald-500'
   261→                )} />
   262→                {isStale ? (
   263→                  <span className="text-amber-700 font-medium">
   264→                    Measurement delayed &middot; Last observation {detail?.last_check_at ? formatDistanceToNow(new Date(detail.last_check_at), { addSuffix: true }) : 'unknown'}
   265→                  </span>
   266→                ) : (
   267→                  <span className="text-emerald-700 font-medium">
   268→                    Measured {detail?.last_check_at ? formatDistanceToNow(new Date(detail.last_check_at), { addSuffix: true }) : 'recently'}
   269→                  </span>
   270→                )}
   271→              </div>
   272→              {detail?.endpoints && (
   273→                <span className="text-[#A1A1AA]">
   274→                  {detail.endpoints.length} endpoint{detail.endpoints.length !== 1 ? 's' : ''} monitored
   275→                </span>
   276→              )}
   277→              {totalChecks !== null && (
   278→                <span className="text-[#A1A1AA]">
   279→                  {totalChecks.toLocaleString()} observations (24h)
   280→                </span>
   281→              )}
   282→            </div>
   283→          </div>
   284→        </motion.div>
   285→      </section>
   286→
   287→      <div className="border-t border-[#E4E4E7]" />
   288→
   289→      {/* ── RELIABILITY SUMMARY ── */}
   290→      <section className="py-8">
   291→        <p className="text-[11px] font-semibold uppercase tracking-[0.1em] text-[#A1A1AA] mb-4">
   292→          Reliability
   293→        </p>
   294→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   295→          <div className="rounded-lg border border-[#E4E4E7] bg-white p-4">
   296→            <span className="text-[11px] font-medium uppercase tracking-[0.05em] text-[#A1A1AA] block mb-2">
   297→              24h uptime
   298→            </span>
   299→            {uptimeValue !== null ? (
   300→              <p className={cn(
   301→                'text-[22px] font-semibold font-mono tracking-tight',
   302→                uptimeValue >= 99.95 ? 'text-emerald-600' : uptimeValue >= 99 ? 'text-amber-600' : 'text-red-600'
   303→              )}>
   304→                {uptimeValue.toFixed(2)}%
   305→              </p>
   306→            ) : (
   307→              <p className="text-sm text-[#A1A1AA]">No data</p>
   308→            )}
   309→          </div>
   310→          <div className="rounded-lg border border-[#E4E4E7] bg-white p-4">
   311→            <span className="text-[11px] font-medium uppercase tracking-[0.05em] text-[#A1A1AA] block mb-2">
   312→              Median latency
   313→            </span>
   314→            {avgLatency !== null ? (
   315→              <p className={cn(
   316→                'text-[22px] font-semibold font-mono tracking-tight',
   317→                avgLatency > 500 ? 'text-red-600' : avgLatency > 200 ? 'text-amber-600' : 'text-[#09090B]'
   318→              )}>
   319→                {Math.round(avgLatency)}<span className="text-[13px] font-normal text-[#A1A1AA] ml-1">ms</span>
   320→              </p>
   321→            ) : (
   322→              <p className="text-sm text-[#A1A1AA]">No data</p>
   323→            )}
   324→          </div>
   325→          <div className="rounded-lg border border-[#E4E4E7] bg-white p-4">
   326→            <span className="text-[11px] font-medium uppercase tracking-[0.05em] text-[#A1A1AA] block mb-2">
   327→              P95 latency
   328→            </span>
   329→            {latestP95 !== null ? (
   330→              <p className={cn(
   331→                'text-[22px] font-semibold font-mono tracking-tight',
   332→                latestP95 > 500 ? 'text-red-600' : latestP95 > 300 ? 'text-amber-600' : 'text-[#09090B]'
   333→              )}>
   334→                {latestP95}<span className="text-[13px] font-normal text-[#A1A1AA] ml-1">ms</span>
   335→              </p>
   336→            ) : (
   337→              <p className="text-sm text-[#A1A1AA]">No data</p>
   338→            )}
   339→          </div>
   340→          <div className="rounded-lg border border-[#E4E4E7] bg-white p-4">
   341→            <span className="text-[11px] font-medium uppercase tracking-[0.05em] text-[#A1A1AA] block mb-2">
   342→              Observations
   343→            </span>
   344→            {totalChecks !== null ? (
   345→              <p className="text-[22px] font-semibold font-mono tracking-tight text-[#09090B]">
   346→                {totalChecks.toLocaleString()}
   347→              </p>
   348→            ) : (
   349→              <p className="text-sm text-[#A1A1AA]">No data</p>
   350→            )}
   351→          </div>
   352→        </div>
   353→      </section>
   354→
   355→      {/* ── LATENCY CHART ── */}
   356→      <section className="pb-8">
   357→        <div className="flex items-center justify-between mb-4">
   358→          <p className="text-[11px] font-semibold uppercase tracking-[0.1em] text-[#A1A1AA]">
   359→            Response Latency
   360→          </p>
   361→          <div className="flex items-center gap-4 text-[11px] text-[#A1A1AA]">
   362→            <div className="flex items-center gap-1.5"><span className="h-2 w-2 rounded-sm bg-[#0891B2]" /> Median</div>
   363→            <div className="flex items-center gap-1.5"><span className="h-2 w-2 rounded-sm bg-[#0891B2]/25" /> P95</div>
   364→          </div>
   365→        </div>
   366→        {chartData.length > 0 ? (
   367→          <div className="rounded-lg border border-[#E4E4E7] p-5 bg-white">
   368→            <div className="flex items-end gap-[2px] h-48">
   369→              {chartData.map((d, idx) => {
   370→                const avgH = (d.latency / maxLatency) * 100;
   371→                const p95H = (d.p95 / maxLatency) * 100;
   372→                return (
   373→                  <div key={idx} className="flex-1 flex flex-col items-center group relative min-w-0">
   374→                    {/* Tooltip */}
   375→                    <div className="absolute -top-10 bg-[#09090B] text-white text-[10px] px-2 py-1 rounded opacity-0 group-hover:opacity-100 transition-opacity pointer-events-none whitespace-nowrap z-10 font-mono">
   376→                      Avg: {d.latency}ms / P95: {d.p95}ms
   377→                    </div>
   378→                    <div className="w-full flex gap-[1px] items-end">
   379→                      <div
   380→                        className="flex-1 rounded-t-sm bg-[#0891B2]/20 transition-all min-h-[1px]"
   381→                        style={{ height: `${Math.max(p95H, 1)}%` }}
   382→                      />
   383→                      <div
   384→                        className={cn(
   385→                          'flex-1 rounded-t-sm transition-all min-h-[1px]',
   386→                          d.latency > 500 ? 'bg-red-400' : d.latency > 300 ? 'bg-amber-400' : 'bg-[#0891B2]'
   387→                        )}
   388→                        style={{ height: `${Math.max(avgH, 1)}%` }}
   389→                      />
   390→                    </div>
   391→                  </div>
   392→                );
   393→              })}
   394→            </div>
   395→            <div className="flex justify-between mt-2 text-[10px] text-[#A1A1AA] font-mono">
   396→              {chartData.filter((_, i) => i % Math.max(Math.floor(chartData.length / 8), 1) === 0).map(d => (
   397→                <span key={d.hour}>{d.hour}</span>
   398→              ))}
   399→            </div>
   400→          </div>
   401→        ) : (
   402→          <div className="rounded-lg border border-[#E4E4E7] bg-white p-12 text-center">
   403→            <p className="text-sm text-[#52525B]">No latency data available yet.</p>
   404→            <p className="text-xs text-[#A1A1AA] mt-1">
   405→              Reliastra has not collected enough observations to display latency trends.
   406→            </p>
   407→          </div>
   408→        )}
   409→      </section>
   410→
   411→      {/* ── MONITORED ENDPOINTS ── */}
   412→      {detail?.endpoints && detail.endpoints.length > 0 && (
   413→        <section className="pb-8">
   414→          <p className="text-[11px] font-semibold uppercase tracking-[0.1em] text-[#A1A1AA] mb-4">
   415→            Monitored Endpoints
   416→          </p>
   417→          <div className="rounded-lg border border-[#E4E4E7] overflow-hidden">
   418→            <table className="w-full text-sm">
   419→              <thead>
   420→                <tr className="bg-[#F8F9FA] border-b border-[#E4E4E7]">
   421→                  <th className="text-left py-3 px-4 text-[11px] font-medium uppercase tracking-[0.05em] text-[#A1A1AA]">Endpoint</th>
   422→                  <th className="text-left py-3 px-4 text-[11px] font-medium uppercase tracking-[0.05em] text-[#A1A1AA]">Regions</th>
   423→                  <th className="text-left py-3 px-4 text-[11px] font-medium uppercase tracking-[0.05em] text-[#A1A1AA]">Status</th>
   424→                  <th className="text-right py-3 px-4 text-[11px] font-medium uppercase tracking-[0.05em] text-[#A1A1AA] hidden sm:table-cell">Last Check</th>
   425→                </tr>
   426→              </thead>
   427→              <tbody>
   428→                {detail.endpoints.map(ep => {
   429→                  const epStatus = statusConfig[ep.health_status] || statusConfig.unknown;
   430→                  return (
   431→                    <tr key={ep.id} className="border-b border-[#F0F0F0] last:border-0 hover:bg-[#FAFAFA] transition-colors">
   432→                      <td className="py-3 px-4 font-mono text-[12px] text-[#09090B] max-w-[300px] truncate">{ep.endpoint_url}</td>
   433→                      <td className="py-3 px-4">
   434→                        <div className="flex flex-wrap gap-1">
   435→                          {ep.regions.map(r => (
