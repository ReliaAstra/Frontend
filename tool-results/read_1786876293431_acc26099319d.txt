    87→    87→function aggregateMedianLatency(
    88→    88→  data: { timestamp: string; latency_ms: number; region: string; dependency_id: string }[]
    89→    89→) {
    90→    90→  if (!data.length) return [];
    91→    91→
    92→    92→  // Group by timestamp, compute median latency per bucket
    93→    93→  const buckets = new Map<string, number[]>();
    94→    94→  for (const d of data) {
    95→    95→    const key = d.timestamp;
    96→    96→    if (!buckets.has(key)) buckets.set(key, []);
    97→    97→    buckets.get(key)!.push(d.latency_ms);
    98→    98→  }
    99→    99→
   100→   100→  return Array.from(buckets.entries())
   101→   101→    .sort(([a], [b]) => a.localeCompare(b))
   102→   102→    .map(([timestamp, latencies]) => {
   103→   103→      const sorted = [...latencies].sort((a, b) => a - b);
   104→   104→      const mid = Math.floor(sorted.length / 2);
   105→   105→      const median =
   106→   106→        sorted.length % 2 !== 0
   107→   107→          ? sorted[mid]
   108→   108→          : (sorted[mid - 1] + sorted[mid]) / 2;
   109→   109→      return { timestamp, latency_ms: Math.round(median * 100) / 100 };
   110→   110→    });
   111→   111→}
