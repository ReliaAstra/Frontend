    73→function uptimeColor(pct: number | null | undefined): string {
    74→  if (pct == null) return "#A1A1AA";

    87→function aggregateMedianLatency(
    88→  data: { timestamp: string; latency_ms: number; region: string; dependency_id: string }[]
    89→) {
    90→  if (!data.length) return [];
    91→
    92→  // Group by timestamp, compute median latency per bucket
    93→  const buckets = new Map<string, number[]>();
    94→  for (const d of data) {
    95→    const key = d.timestamp;
    96→    if (!buckets.has(key)) buckets.set(key, []);
    97→    buckets.get(key)!.push(d.latency_ms);
    98→  }
    99→
   100→  return Array.from(buckets.entries())
   101→    .sort(([a], [b]) => a.localeCompare(b))
   102→    .map(([timestamp, latencies]) => {
   103→      const sorted = [...latencies].sort((a, b) => a - b);
   104→      const mid = Math.floor(sorted.length / 2);
   105→      const median =
   106→        sorted.length % 2 !== 0
   107→          ? sorted[mid]
   108→          : (sorted[mid - 1] + sorted[mid]) / 2;
   109→      return { timestamp, latency_ms: Math.round(median * 100) / 100 };
   110→    });
   111→}
