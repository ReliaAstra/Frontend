    48→    48→type StatusBadgeType = "operational" | "degraded" | "down" | "unknown";
    49→    49→
    50→    50→function deriveStatus(dep: Dependency): StatusBadgeType {
    51→    51→  if (!dep.is_active) return "unknown";
    52→    52→  // If no recent check info, treat as unknown. In a real app you'd check last check result.
    53→    53→  return "operational";
    54→    54→}

    83→    83→  if (pct === null) return "#52525B";
    84→    84→  if (pct >= 99) return "#16A34A";
