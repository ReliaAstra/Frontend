    38→    38→function formatDuration(seconds: number): string {
    39→    39→  if (seconds < 60) return `${seconds}s`;
    40→    40→  if (seconds < 3600) return `${Math.floor(seconds / 60)}m ${seconds % 60}s`;
    41→    41→  const h = Math.floor(seconds / 3600);
    42→    42→  const m = Math.floor((seconds % 3600) / 60);
    43→    43→  return `${h}h ${m}m`;
    44→    44→}
